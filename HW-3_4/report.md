# Задача 1. Hello, World! 

Вывод с помощью puts
```
.text
.globl main

main:
	pushq %rbp
	movq %rsp, %rbp

	leaq msg(%rip), %rdi # загружаем нашу строку из .rodata
	call puts # используем puts для вывода

	movq $0, %rax # возвращаемое значение из main

	leave
	retq


.section .rodata
msg:
	.string "Hello, world!"
  ```

С использованием syscalls

  ```.text
.globl main

main:
	pushq %rbp
	movq %rsp, %rbp

	movq $1, %rax # номер write syscall
	movq $1, %rdi # stdout
	movq $msg, %rsi # указатель на строку
	movq $14, %rdx # длина строки
	syscall 

	movq $60, %rax # номер exit syscall
	movq $0, %rdi 

	syscall 



.section .data
msg:
	.string "Hello, world!\n"
  ```

# Задача 2. Сложение

```
.text
.globl main
.globl add


add:
	addq %rsi, %rdi # %rdi = %rdi + %rsi 
	movq %rdi, %rax # возвращаемое значение должно быть в %rax согласно abi
	ret


main:
	pushq %rbp
	movq %rsp, %rbp

	movq $15, %rdi # первый аргумент, используем %rdi согласно abi
	movq $10, %rsi # второй аргумент, используем %rsi согласно abi

	call add

	leaq msg(%rip), %rdi # форматная строка для printf
	movq %rax, %rsi # первый аргумент для форматной строки
	movb $0, %al
	call printf # вызываем printf

	movq $0, %rax # возвращаемое значение из main
	leave
	ret


.section .data
msg:
	.string "Result: %d\n"
```

# Задача 3. Fibonacci

```
.text
.globl main
.globl fibonacci

fibonacci:
	cmpq $0, %rdi # проверяем, что переданный аргумент не равен нулю
	je .equal # если равен, но прыгаем на equal и возвращаем 0

	movq $0, %rdx # начальное значение, которое использует для рассчёта числа
	movq $1, %rax  # начальное значение, которое использует для рассчёта числа

	movq $2, %rcx # этот регистр используем как счётчик, начинаем с 2, т.к. первые два числа уже посчитаны 0, 1

.loop_start:
	cmpq %rdi, %rcx # сравнием переданное число со счётчик
	jg .loop_end # цикл идёт пока счётчик не станет больше нашего числа
	movq %rax, %r8 # используем %r8 как промежуточное значение
	add %rdx, %rax # считаем следующее число фибоначчи 
	movq %r8, %rdx # # сохраняем прожуточное число обратно в регистр, будем его использовать в следующих вычислениях

	incq %rcx # увеличиваем счётчик
	jmp .loop_start



.loop_end:
	ret

.equal:
	movq $0, %rax
	ret



main:
	pushq %rbp
	movq %rsp, %rbp

	movq $input, %rdi # указатель на форматную строку
	movq $num, %rsi # куда сохранить
	movq $0, %rax # целочисленное число
	call scanf
	movq num(%rip), %rdi # сохраняем переданное число в %rdi, т.к. это первый аргумент нашей функции fib, согласно abi

	call fibonacci # вызываем нашу функцию

	leaq msg(%rip), %rdi
	movq %rax, %rsi
	movb $0, %al
	call printf

	movq $0, %rax
	movq %rbp, %rsp
	pop %rbp
	ret


.section .data
msg:
	.string "Result: %d\n"
input:
	.string "%d"
num:
	.int 0
```

# Задача 4. Binary Bomb

Сначала сделал `objdump -d bomb_hard > objdump_output.txt`, чтобы посмотреть какие вообще функции есть. Были выделены основные функции: `main`, `phase_1`, `phase_2`, `phase_3`, `phase_4`, `secret_phase`.

Начал с функции `main`, дальше уже смотрел дизассемблер в `gdb`, т.к. там есть подстветка и удобно смотреть на какой адрес памяти происходит прыжок. 


## Фаза 1
При запуске программы сначала нам выводится: `[Phase 1] Enter a string:`, следовательно для прохождения первой фазы нужно найти строку. Используем команду `disassemble phase_1` в `gdb`. Выделяем следующие команды:
```
   0x00000000004011b8 <+8>:	mov    %rdi,-0x8(%rbp)
   0x00000000004011bc <+12>:	mov    -0x8(%rbp),%rdi
   0x00000000004011c0 <+16>:	mov    $0x402069,%esi
   0x00000000004011c5 <+21>:	call   0x401060 <strcmp@plt>
```

Видим, что происходит сравнение строк. Один из аргументов это строка по адресу `$0x402069`. Её мы можем прочитать с помощью `gdb` команды `x/s 0x402069`. Получаем следующую строку: `Rust is blazingly fast and memory-efficient`. Это и есть решение для первой фазы.

## Фаза 2

Также делаем `disassemble phase_2`. Сначала делается `scanf` со следующей форматной строкой: `"%d %d %d %d %d %d"` (получил с помощью `x/s 0x4020a6`). Следовательно мы должны передать 6 чисел через пробел. Числа будут хранится на стеке вот в такоем виде:

| Index | Stack Address |
|-------|---------------|
| 5     | -0xc(%rbp)    |
| 4     | -0x10(%rbp)   |
| 3     | -0x14(%rbp)   |
| 2     | -0x18(%rbp)   |
| 1     | -0x1c(%rbp)   |
| 0     | -0x20(%rbp)   |

Одна из первых проверок:
```   
0x0000000000401254 <+100>:	cmpl   $0x1,-0x20(%rbp)
0x0000000000401258 <+104>:	je     0x401263 <phase_2+115>
```

Из неё делается вывод, что первое наше переданное число должно быть **1**. Прыгаем на `phase_2+115`. 

Там мы делаем следующие операции:
```
0x0000000000401263 <+115>:	movl   $0x1,-0x24(%rbp) - сохраняем единицу
0x000000000040126a <+122>:	cmpl   $0x6,-0x24(%rbp) - сравниваем 1 и 6
0x000000000040126e <+126>:	jge    0x4012ab <phase_2+187> если true, то прыгаем иначе продолжаем
```

Нам нужен прыжок, т.к. там конец функции, но пока мы этого сделать не можем. 

Дальше мы грузим ранее загруженную 1 в `-0x24(%rbp)` в `%rax` и делаем следующую операцию: `mov    -0x20(%rbp,%rax,4),%eax` - с помощью этой операции мы получаем число из нашего input, в данном случае наше второе число и сохраняем его в `%eax`.

В `%exc` мы загружаем наш счётчик и вычитаем из него единицу. Дальше мы снова берём число из нашего input, но уже предыдущее. И вот это предыдущее число мы смещаем влево на один байт (делаем умножение на 2). 

То есть, что у нас получается. Первое число должно быть **1**, потом мы берём это число, сдвигаем влево и сравниваем со следующим числом. Следовательно следующее число должно быть в два раза больше. Следовательно это 2. Точно также из с остальными числами.

Получается следующая последовательность: `1 2 4 8 16 32`. Это отве для второй фазы

## Фаза 3

Снова делаем `disassemble phase_3` и видим `sscanf`. Смотрим какая форматная строка используется (`x/s 0x4020b2`): `%d %d`. Должны передать два числа. 

В `-0xc(%rbp)` первое число, в `-0x10(%rbp)` хранится второе число. Первое число мы загружаем в `%eax`, проверяем, что оно <= 5. Также первое число мы сохраняем в `-0x18(%rbp)`. Дальше мы делаем следующие операции:

```
0x000000000040131c <+76>:	mov    0x402008(,%rax,8),%rax
0x0000000000401324 <+84>:	jmp    *%rax
```

То есть мы читаем данные из памяти и прыгаем адрес, который указан в этом участке памяти. Можно сделать вывод, что первое число задаёт смещение. Посмотрим, что хранится по адресу 0x402008 (`x/16gx 0x402008`). Видимо следующее:

```

0x402038:	0x4f4f42202a2a2a0a	0x2a2a2a202121214d
0x402048:	0x6f6220656854000a	0x6f6c70786520626d
0x402058:	0x7972542021646564	0x0a2e6e6961676120
```

Нам бы хотелось прыгнуть на `0x000000000040133d`. Следовательно нам нужно передать **1**.

Дальше происходит уже сравнение второго числа:

```
0x000000000040133d <+109>:	cmpl   $0x6b,-0x10(%rbp)
```

Следовательно нам нужно, чтобы второе число было 0x6b => 107 десятичной системе.

Ответ для этой фазы: 1 107

# Фаза 4

Должны передать следующую  строку: `"%d %19s"` подходящая под этот формат. 

Тут вызывается функция `fib`, которая считает число фибоначчи и результат сравнивается с нашим переданным числов. Сравнивается с числом 55. Следовательно мы должны передать число 10, т.к. это число даёт 55. 

```
   0x0000000000401498 <+88>:	mov    -0xc(%rbp),%edi # загружаем наше число как аргумент функции
```
